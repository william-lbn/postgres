import importlib.util,json,os,re,signal,subprocess,time,unittest
from pathlib import Path
spec=importlib.util.spec_from_file_location('dsm18_prepared',Path(__file__).with_name('test_prepared18.py'));prep=importlib.util.module_from_spec(spec);spec.loader.exec_module(prep)
conn=prep.conn;region=prep.region
# 已定位原pgstat在startup恢复统计：in-place256KiB+固定首个外部DSA1MiB。
BASELINE=1048576
class Cluster(prep.Cluster):
 def __init__(self,budget='64kB',method='posix',main='0',**kw):
  super().__init__(injection=True,**kw)
  self.expected_baseline=0 if main!='0'else BASELINE;self.expected_segments=0 if main!='0'else 1
  allowance={'0':0,'8kB':8192,'16kB':16384,'64kB':65536}[budget];self.configured_cap=0 if budget=='0'else self.expected_baseline+allowance
  with (self.data/'postgresql.conf').open('a')as f:f.write(f"dynamic_shared_buffers_dsm_budget='{self.configured_cap//1024}kB'\ndynamic_shared_memory_type='{method}'\nmin_dynamic_shared_memory='{main}'\n")
class DSM(unittest.TestCase):
 wait=conn.Connection.wait;gate=conn.Connection.gate;blocked=conn.Connection.blocked;fact=conn.Connection.fact
 def cluster(self,**kw):
  p=Cluster(**kw);self.addCleanup(p.close);p.start();p.sql('CREATE ROLE work18_user LOGIN')
  if p.mode is not None:
   initial=self.snap(p);self.assertEqual(int(initial['reserved_upper_bytes']),p.expected_baseline);self.assertEqual(initial['external_segments'],p.expected_segments);self.assertEqual(initial['pending_creations'],0);self.assertEqual(initial['unknown_segments'],0);self.assertEqual(int(initial['external_budget_bytes']),p.configured_cap)
   p.sql('SELECT test_dsg_dsm18_pgstat_area()');until=time.monotonic()+2;dump=''
   while time.monotonic()<until:
    log=(p.data/'logs/pg.log').read_text();dump=log.split('PG18_DSM_TEST_PGSTAT_AREA_BEGIN')[-1].split('PG18_DSM_TEST_PGSTAT_AREA_END')[0]
    if 'total_segment_size: 1310720'in dump:break
    time.sleep(.01)
   self.assertIn('total_segment_size: 1310720',dump);self.fact(p.root.name+'-exact-native-baseline',{'actual_total_snapshot':initial,'configured_absolute_cap':p.configured_cap,'required_fixed_baseline':p.expected_baseline,'actual_pgstat_area_dump':dump,'unproven_dynamic_baseline_not_accepted':True})
  return p
 def snap(self,p):
  s=json.loads(p.sql('SELECT test_dsg_dsm18_snapshot()').stdout)
  if s['limit_enforced']and s['accounting_valid']:self.assertLessEqual(int(s['reserved_upper_bytes']),int(s['external_budget_bytes']))
  return s
 def create(self,p,size=8192,pin=True):return json.loads(p.sql(f'SELECT test_dsg_dsm18_create({size},{str(pin).lower()})').stdout)
 def touch(self,p,h,release=False):return json.loads(p.sql(f'SELECT test_dsg_dsm18_touch({h},{str(release).lower()})').stdout)
 def zero(self,p):return self.wait(p,lambda s:int(s['reserved_upper_bytes'])==p.expected_baseline and s['external_segments']==p.expected_segments and s['pending_creations']==0 and s['unknown_segments']==0)
 def attach(self,p,name,action):p.sql("SELECT injection_points_attach('"+name+"','"+action+"')")
 def detach(self,p,name):p.sql("SELECT injection_points_detach('"+name+"')")
 def wake(self,p,name):p.sql("SELECT injection_points_wakeup('"+name+"')",check=False)
 def async_sql(self,p,sql,app):
  e=region.env();e['PGAPPNAME']=app;c=subprocess.Popen([str(region.BIN/'psql'),'-h',str(p.sock),'-p','55438','-U',region.USER,'-d','postgres','-XqAt','-v','ON_ERROR_STOP=1','-c',sql],env=e,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True);self.addCleanup(lambda:c.kill()if c.poll()is None else None);return c
 def test_PG18_U_DSM1_001_defaultoff_original_session_ABI_and_positive_unsupported(self):
  off=self.cluster(mode=None,gate=False,admission=False,budget='0');bad=off.sql('SELECT test_dsg_dsm18_snapshot()',check=False);self.assertNotEqual(bad.returncode,0);self.assertIn('registered running server',bad.stderr)
  p=Cluster(mode=None,gate=False,admission=False,budget='64kB');self.addCleanup(p.close);e=region.env();bad=subprocess.run([str(region.BIN/'postgres'),'-D',str(p.data)],env=e,capture_output=True,text=True,timeout=10);self.assertNotEqual(bad.returncode,0);self.assertIn('budget requires managed native buffer profile',bad.stderr)
  p=self.cluster(budget='0');s=self.snap(p);self.assertEqual((s['api'],s['prepared_api'],s['connection_api'],s['work_api']),(1,1,1,2));self.assertFalse(s['limit_enforced']);self.assertTrue(s['accounting_valid']);self.assertFalse(s['normal_permit']);v=self.create(p,8192);tracked=self.snap(p);self.assertEqual(int(tracked['reserved_upper_bytes']),p.expected_baseline+8192);self.touch(p,v['handle'],True);self.zero(p);self.gate(p,True)
  self.assertEqual(p.sql("SELECT has_function_privilege('work18_user','test_dsg_dsm18_snapshot()','EXECUTE')").stdout.strip(),'f');p.sql('GRANT EXECUTE ON FUNCTION test_dsg_dsm18_snapshot() TO work18_user');p.sql(f"CREATE FUNCTION dsm18_definer() RETURNS text AS '{region.LIB}','test_dsg_dsm18_snapshot' LANGUAGE C SECURITY DEFINER")
  for sql in ['SET ROLE work18_user;SELECT test_dsg_dsm18_snapshot()','SET SESSION AUTHORIZATION work18_user;SELECT dsm18_definer()']:
   bad=p.sql(sql,check=False);self.assertNotEqual(bad.returncode,0);self.assertIn('original administrator',bad.stderr)

 def test_PG18_S_DSM1_002_preOS_pending_reservation_blocks_concurrent_creator(self):
